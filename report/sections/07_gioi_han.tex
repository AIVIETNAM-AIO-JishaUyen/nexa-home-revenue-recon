\section{Giới hạn và bước tiếp theo}

\subsection{Giới hạn của kết luận}

\begin{itemize}
  \item \textbf{Chỉ kết luận điều dữ liệu cho thấy.} Dữ liệu cho biết NX057 thiếu bút toán ngày 21/11 và được ghi bù ngày 02/12. Bút toán ghi bù chỉ ghi ``lỗi interface'' (lỗi ở bước chuyển dữ liệu từ POS sang sổ cái), không cho biết lỗi gì. Dữ liệu cũng không cho biết ADJ-0925 ghi cho việc gì. Hai câu này phải hỏi IT và phòng Tài chính; nhóm không đoán.
  \item \textbf{Chỉ có dữ liệu một năm.} Lỗi chuyển dữ liệu (NX057) và lỗi xuất trùng (NX007) mỗi loại chỉ thấy một lần trong năm 2025. Chưa biết hai lỗi này lặp lại thường xuyên đến đâu.
  \item \textbf{Số làm tròn có ba phiên bản.} Nhóm tính +\SoLamTron{} USD theo định nghĩa slide 34, khớp slide 32. Dashboard mẫu của mentor ra +2,33 USD, vì cộng các cột USD đã làm tròn từng dòng (Phụ lục~D). Chênh chưa tới 2 USD này không đổi kết luận.
  \item \textbf{Một số quy ước chờ mentor xác nhận}, ví dụ thứ tự các bước cầu nối, cách đặt ADJ-0925 (nhóm đã ghi sẵn 8 câu hỏi xin mentor trong file \texttt{plans/backlog.md} của dự án). Nếu mentor trả lời khác, số từng khoản có thể đổi, nhưng tổng chênh lệch không đổi.
  \item \textbf{Việc chạy lại cho tháng mới chỉ mới thử bằng dữ liệu giả, ở một thư mục khác} (mục~II.8). Chưa thử trên máy thật thứ hai. Phần Power BI chưa thử với tháng mới, vì file dữ liệu cho Power BI chỉ có năm 2025.
  \item \textbf{Dashboard chưa có nút ``Đặt lại'' bộ lọc}, và chưa tô tháng đang chọn trên biểu đồ 12 tháng (mục~IV.6).
  \item \textbf{Flash không có số theo cửa hàng.} Vì vậy chỉ so được Flash với sổ cái ở mức tháng, kênh, thị trường. Ở mức cửa hàng, nhóm phải so POS với sổ cái.
\end{itemize}

\subsection{Rủi ro đề không nhắc tới}

\begin{itemize}
  \item \textbf{Hàng trả sau ngày lấy dữ liệu.} Dữ liệu chỉ lấy đến 15/01/2026. Đã có 7 đơn giao năm 2025 bị trả trong tháng 01/2026 (nhật ký EX-13); các đơn giao cuối tháng 12 vẫn có thể bị trả tiếp. Hàng trả ghi giảm doanh thu vào ngày nhận, nên không đổi số năm 2025 nhưng làm giảm doanh thu tháng 01/2026.
  \item \textbf{Sổ cái tháng 11 đã khóa mà vẫn thiếu doanh thu.} Lỗi NX057 được sửa bằng bút toán ghi bù tháng 12, nên số sổ cái tháng 11 đã chốt thấp hơn thực tế 7.227,56 USD. Đề xuất 1 nhằm chặn đúng việc này.
  \item \textbf{Mã cửa hàng khác nhau giữa POS và sổ cái.} Trong bảng danh mục cửa hàng, mã POS cũ có trạng thái ``Restructured'' (tái cấu trúc), còn mã POS mới vẫn gắn cùng mã sổ cái. Vì vậy một số mã sổ cái có nhiều mã POS (ví dụ NX003 có 2 mã, NX007 có 3 mã). Năm 2025 mỗi mã sổ cái chỉ có một mã POS còn bán, nên chưa gây lệch. Nếu bảng danh mục cửa hàng không cập nhật kịp khi mở hoặc chuyển cửa hàng, doanh thu POS sẽ không ghép được với sổ cái. Phép kiểm tra số 9 (dòng POS có mã cửa hàng không có trong danh mục) sẽ báo.
\end{itemize}

\subsection{Bước tiếp theo}

\begin{enumerate}
  \item Hỏi mentor 8 câu hỏi đã ghi sẵn. Nếu câu trả lời khác quy ước thì sửa file quy ước và chạy lại (khoảng 3 đến 5 phút).
  \item Hỏi phòng Tài chính chứng từ của ADJ-0925, hỏi IT nguyên nhân lỗi NX057 và lỗi xuất trùng.
  \item Chạy lại toàn bộ trên một máy thứ hai trước khi nộp.
\end{enumerate}
